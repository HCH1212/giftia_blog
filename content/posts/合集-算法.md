## LRU
```go
type LRUCache struct {
    len int
    capacity int
    cache map[int]*Node
    head, tail *Node
}

type Node struct {
    key, value int
    prev, next *Node
}


func Constructor(capacity int) LRUCache {
    head, tail := &Node{}, &Node{}
    head.next = tail
    tail.prev = head
    return LRUCache{
        capacity: capacity,
        cache: make(map[int]*Node),
        head: head,
        tail: tail,
    }
}


func (this *LRUCache) Get(key int) int {
    node, ok := this.cache[key]
    if !ok {
        return -1
    }
    deleteNode(node)
    addHead(this.head, node)
    return node.value
}


func (this *LRUCache) Put(key int, value int)  {
    node, ok := this.cache[key]
    if ok {
        node.value = value
        deleteNode(node)
        addHead(this.head, node)
        return
    }
    node = &Node{key:key, value:value}
    if this.len < this.capacity {
        addHead(this.head, node)
        this.cache[key] = node
        this.len++
    } else {
        lastNode := this.tail.prev
        delete(this.cache, lastNode.key)
        deleteNode(lastNode)
        addHead(this.head, node)
        this.cache[key] = node
    }
}

func addHead(head, node *Node) {
    head.next.prev = node
    node.next = head.next
    head.next = node
    node.prev = head
}

func deleteNode(node *Node) {
    node.prev.next = node.next
    node.next.prev = node.prev
}
```
> 双向链表，记得同步更新链表和map
